{{/* Fails the build on content that breaks a URL, expiry, or link invariant. Renders nothing. Called once per build via partialCached from baseof.html. These rules were previously enforced only by the new-* skills at creation time, which cannot catch a hand-written file or an edit made in review (issues #56, #61). */}} {{/* Obsidian wikilinks do not fail — Goldmark renders [[Target]] as literal brackets, so the build exits 0 and the page publishes with visible markup. The notes arrived from an Obsidian vault and 227 wikilinks were converted on import; anything editing this content in Obsidian will offer to put them back, and neither of Obsidian's link modes can produce a working link here, because a URL is Hugo's normalization of the title, not the vault path. Checked across every page with a file, not just the dated sections, so it also covers governance documents and section _index.md files. Frontmatter is out of scope: RawContent excludes it. */}} {{- range site.Pages -}} {{- if .File -}} {{- if strings.Contains .RawContent "[[" -}} {{- errorf "%s: contains an Obsidian wikilink ([[...]]) — Hugo publishes these as literal brackets. Write a root-relative link instead, e.g. [Label](/notes/some-slug/)." .File.Path -}} {{- end -}} {{- end -}} {{- end -}} {{/* #73: cross-document citations carry a heading fragment (/governance/bylaws/#article-iv--officers). Goldmark derives those anchors from the heading text, so editing a heading silently breaks every inbound citation — no error, and the link still resolves to the top of the page. That is the same damage #56 gated for the slug half of the same URL. Pages are resolved by RelPermalink rather than content path: the two now agree for dated posts, but not for governance or section pages. Section pages are included: one of the citations lives in minutes/_index.md. Limitation: .Fragments knows content headings only, so a fragment pointing at an id emitted by a template would report as missing. None do today. */}} {{- $byURL := dict -}} {{- range site.Pages -}} {{- $byURL = merge $byURL (dict .RelPermalink .) -}} {{- end -}} {{- range site.Pages -}} {{- if .File -}} {{- $file := .File.Path -}} {{- range findRESubmatch `\]\((/[^)#\s]*)#([^)\s]+)\)` .RawContent -}} {{- $path := index . 1 -}} {{- $frag := index . 2 -}} {{- $key := cond (hasSuffix $path "/") $path (printf "%s/" $path) -}} {{- with index $byURL $key -}} {{- if not (in .Fragments.Identifiers $frag) -}} {{- errorf "%s: link to %s#%s — %s has no heading with that anchor. Goldmark derives anchors from heading text, so retitling a heading breaks every citation of it." $file $path $frag $path -}} {{- end -}} {{- else -}} {{- errorf "%s: link to %s#%s — no page at %s." $file $path $frag $path -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- range $section := slice "notices" "minutes" "news" "notes" "governance" -}} {{- $seen := dict -}} {{- with site.GetPage (printf "/%s" $section) -}} {{/* Sorted by file path so a duplicate URL always names the same file as the one to fix (#78) — the first alphabetically owns it, the later one is reported. No section date-prefixes its filenames, so this is alphabetical order. .ByDate would not do: nothing stops two posts in a section sharing a date, and a date carries no time, so which file got named would be free to drift between builds. */}} {{- range sort .RegularPages "File.Path" -}} {{- $file := .File.Path -}} {{/* #56: nothing in frontmatter pins a URL any more — it is the title as Hugo normalizes it, and the filename is the same normalization written by hand. Requiring the two to agree is what catches a retitle that was not accompanied by a rename, which is the edit that silently moves a published URL. It also keeps the file findable from the URL. */}} {{- $seg := path.Base .RelPermalink -}} {{/* #113: governance is deliberately absent from [permalinks], so its URLs come from the filename and a prose retitle cannot move them. That is the right property for the four most heavily cited documents on the site — the Bylaws alone are cited from the minutes, the notice, the Standing Rules and the Officers page, several with heading fragments. It also makes the check below vacuous there, since the URL segment IS the filename. So governance is checked the other way round: the normalized title against the filename. A retitle that skips the rename fails the build instead of silently orphaning the page from its own name. */}} {{- if eq $section "governance" -}} {{- $want := urlize .Title -}} {{- if ne .File.ContentBaseName $want -}} {{- errorf "%s: title %q normalizes to %q but the filename is %q — governance URLs come from the filename, so either restore the title or rename the file to %s.md. Renaming moves a URL that other documents cite: add the old path under aliases: and check the citations." $file .Title $want .File.ContentBaseName $want -}} {{- end -}} {{- else -}} {{- if ne .File.ContentBaseName $seg -}} {{- errorf "%s: filename does not match its URL segment %q — Hugo derives the URL from the title, so rename the file to %s.md (note that Hugo drops & entirely; write \"and\" if you want it in the URL)." $file $seg $seg -}} {{- end -}} {{- end -}} {{- with index $seen $seg -}} {{- errorf "%s: URL /%s/%s/ is already taken by the earlier post %s — retitle this post." $file $section $seg . -}} {{- end -}} {{- $seen = merge $seen (dict $seg $file) -}} {{/* #112: notice_type and meeting_type carry a controlled vocabulary that lived only in README and the new-notice / new-minutes skills. Nothing resolved them, so a typo published as a title-cased heading and exited 0 — notice-meta.html renders whatever it is given. Same reasoning as #56 and #61: a rule enforced only at creation time cannot catch a hand-written file or an edit made in review. */}} {{- if and (eq $section "notices") .Params.notice_type -}} {{- if not (in (slice "annual-meeting" "special-meeting" "previous-notice" "bylaw-amendment") .Params.notice_type) -}} {{- errorf "%s: notice_type %q is not one of the four documented values (annual-meeting, special-meeting, previous-notice, bylaw-amendment)" $file .Params.notice_type -}} {{- end -}} {{- end -}} {{- if and (eq $section "minutes") .Params.meeting_type -}} {{- if not (in (slice "annual" "special" "regular") .Params.meeting_type) -}} {{- errorf "%s: meeting_type %q is not one of the three documented values (annual, special, regular)" $file .Params.meeting_type -}} {{- end -}} {{- end -}} {{/* #61: expires is the first day hidden, so it must clear the meeting. */}} {{- if and .Params.expires .Params.meeting_date -}} {{- if not (gt (time .Params.expires) (time .Params.meeting_date)) -}} {{- errorf "%s: expires %s is not after meeting_date %s — expires is the first day the notice is hidden" $file .Params.expires .Params.meeting_date -}} {{- end -}} {{- end -}} {{/* #75: approval is a pair. `approved: true` with no approved_on reads "Approved" with no date, which under RONR asserts a status the record cannot substantiate — the date identifies the meeting that adopted it. */}} {{- if and (eq $section "minutes") .Params.approved (not .Params.approved_on) -}} {{- errorf "%s: approved: true with no approved_on — both move together; set approved_on to the date the minutes were adopted" $file -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
